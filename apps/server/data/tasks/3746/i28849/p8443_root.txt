function onBodyLoad()
{
	var lang = getBrowserLanguage();
	if ( lang.indexOf("cn")!=-1 || lang.indexOf("CN")!=-1)
	{
		if($.browser.msie && $.browser.version <= 8.0)
		{
			window.location = "/html/nopen.html";
			return;
		}
		window.location = "html/login.html";
	}
	else
	{
		if($.browser.msie && $.browser.version <= 8.0)
		{
			window.location = "/html/nopen_en.html";
			return;
		}
		window.location = "html/login_en.html";
	}

}

function getBrowserLanguage()
{
	if ($.browser.msie)
	{
		return window.navigator.browserLanguage;
	}

	// netscape
	return window.navigator.language;
}

</script>
